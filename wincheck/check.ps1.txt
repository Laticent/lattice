# check.ps1: install the real installer, launch it, run run.ps1 (UI Automation + real OS input).
$ErrorActionPreference = 'Continue'
$root = Split-Path -Parent $PSScriptRoot
$out = Join-Path $root 'wincheck-results'
New-Item -ItemType Directory -Force $out | Out-Null
$log = Join-Path $out 'log.txt'
function Say($m) { Write-Host $m; Add-Content $log $m }

# Compiled by Windows PowerShell 5.1 (.NET Framework), which is what win.ps1 runs on: a DLL
# from PowerShell 7 (.NET 8) does not load there ("Bad IL format").
$cs = Join-Path $PSScriptRoot 'Win.cs'; $dll = Join-Path $PSScriptRoot 'Win.dll'
powershell -NoProfile -Command "Add-Type -TypeDefinition (Get-Content '$cs' -Raw) -ReferencedAssemblies System.Windows.Forms, System.Drawing -OutputAssembly '$dll' -OutputType Library"
$win = Join-Path $PSScriptRoot 'win.ps1'
function W { powershell -NoProfile -ExecutionPolicy Bypass -File $win @args }
Say "Win.dll: $(Test-Path $dll); focus test: $(W windows | Measure-Object -Line | Select-Object -ExpandProperty Lines) visible windows"

$exe = Get-ChildItem (Join-Path $root 'installer') -Recurse -Filter '*setup.exe' | Select-Object -First 1
Say "installer: $($exe.FullName) sha256=$((Get-FileHash $exe.FullName).Hash)"
Start-Process $exe.FullName -ArgumentList '/S' -Wait
$app = Get-ChildItem $env:LOCALAPPDATA -Recurse -Filter 'lattice-studio.exe' -ErrorAction SilentlyContinue | Select-Object -First 1
if (-not $app) { Say 'FAIL install: lattice-studio.exe not found under LOCALAPPDATA'; exit 0 }
Say "installed: $($app.FullName)"
$wv = (Get-ItemProperty 'HKLM:\SOFTWARE\WOW6432Node\Microsoft\EdgeUpdate\Clients\{F3017226-FE2A-4295-8BDF-00C3A9A7E4C5}' -ErrorAction SilentlyContinue).pv
Say "WebView2 runtime: $wv; Windows: $((Get-CimInstance Win32_OperatingSystem).Caption) $((Get-CimInstance Win32_OperatingSystem).Version)"
Say "default browser: $((Get-ItemProperty 'HKCU:\Software\Microsoft\Windows\Shell\Associations\UrlAssociations\https\UserChoice' -ErrorAction SilentlyContinue).ProgId)"

Start-Process $app.FullName
Start-Sleep 8
Say "--- processes"
Get-CimInstance Win32_Process | Where-Object { $_.Name -match 'lattice' -or ($_.CommandLine -match 'embedded-browser-webview=1' -and $_.CommandLine -notmatch '--type=') } | ForEach-Object { Say ("{0} {1} :: {2}" -f $_.ProcessId, $_.Name, $_.CommandLine) }
Say "--- windows"
Say ((W windows) -join "`n")
W shot (Join-Path $out '00-after-launch.png')
Say "screenshot written: $(Test-Path (Join-Path $out '00-after-launch.png'))"
Copy-Item (Join-Path $PSScriptRoot 'run.ps1.txt') (Join-Path $PSScriptRoot 'run.ps1')
powershell -NoProfile -ExecutionPolicy Bypass -File (Join-Path $PSScriptRoot 'run.ps1') -Out $out 2>&1 | Tee-Object -FilePath (Join-Path $out 'run-output.txt')
Say 'done'
